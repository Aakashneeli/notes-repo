A small response schema, before your own

from typing import Literal
from pydantic import BaseModel
class Receipt(BaseModel):
    state: Literal["accepted"]

print(Receipt(state="accepted").model_dump())
# {'state': 'accepted'}
# Receipt(state="finished") raises ValidationError.

class Receipt(BaseModel) inherits Pydantic validation. The annotated field is required because it has no default. Constructing the object checks the value; model_dump() returns an ordinary dictionary.
